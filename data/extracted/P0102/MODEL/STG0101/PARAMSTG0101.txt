// フォグの情報
//   1つ目:フォグの種類
//           0:NONE 3:EXP2
//           1:LIN  4:REVEXP
//           2:EXP  5:REVEXP2
//   2つ目:フォグの赤(0～255)
//   3つ目:フォグの緑(0～255)
//   4つ目:フォグの青(0～255)
//   5つ目:フォグの掛かり始める距離
//   6つ目:フォグで見えなくなる距離
#FOG 1 45 55 194 -1000 10000

// スクリーンフィルターのパラメータ
//   1つ目:フィルターの種類
//           1:ノイズフィルタ
//
//   2つ目以降はフィルタによって違う
//       ノイズフィルタの場合
//           2つ目:ぼかしの解像度(0%～100%)
//           3つ目:ぼかしの濃さ(0%～100%)
//           4つ目:乗算の濃さ(0%～100%)
//           5つ目:ノイズアニメーションの間隔(0フレーム～100フレーム)
//           6つ目:ぼかしの赤(0～255)
//           7つ目:ぼかしの緑(0～255)
//           8つ目:ぼかしの青(0～255)
#FILTER 1 10 30 0 0 55 240 145

// 影の色( 赤・緑・青・透明度の順 (0～255) )
#SHADOW_COLOR 0 0 5 200

// 遠景ブルーム
#FAR_BLOOM_ENABLE 0

// 遠景ブルーム色( R G B A )
#FAR_BLOOM_COLOR 0 153 251 155

// 敷居値ブルーム
#THRESHOLD_BLOOM_ENABLE 1

// ブルーム敷居値色( R G B A )
#THRESHOLD_COLOR 250 250 250

// ステージで使用するサウンドエフェクト名
#SOUNDEFFECT  "EFFECT_SKELTER"

// ステージの音量減衰の単位距離
#SOUNDUNITDISTANCE  100.0

#BACK_COLOR 0 0 0 

// ファイルの終端
#EOF
